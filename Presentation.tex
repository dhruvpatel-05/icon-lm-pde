\section{Motivation and the ICON-LM idea}

%----- Slide 2: Motivation -----------------------------------
\begin{frame}{Why operator learning needs a new paradigm}
  \begin{columns}[T,onlytextwidth]
    \column{0.55\textwidth}
      \textbf{Classic operator learning} (DeepONet, FNO, \dots)
      \begin{itemize}
        \item Train \emph{one} network to approximate \emph{one} operator $T: c \mapsto u$.
        \item New PDE $\Rightarrow$ new dataset $\Rightarrow$ re-train.
        \item No way to inject what we \emph{know} about the equation.
      \end{itemize}
      \vspace{0.3em}
      \textbf{The aspiration: a foundation model for DEs}
      \begin{itemize}
        \item \emph{Train once, apply many.}
        \item Learn operators \emph{at inference time} from a prompt.
        \item Directly inspired by in-context learning in GPT-style LMs.
      \end{itemize}
    \column{0.42\textwidth}
      \begin{tikzpicture}[scale=0.8,transform shape]
        \tikzset{
          tok/.style={draw,rounded corners,minimum width=1.4cm,minimum height=0.6cm,fill=iconlight,font=\scriptsize},
          qtok/.style={draw,rounded corners,minimum width=1.4cm,minimum height=0.6cm,fill=iconaccent!25,font=\scriptsize}
        }
        \node[tok] (e1) at (0,0)   {$(c_1,u_1)$};
        \node[tok] (e2) at (1.7,0) {$(c_2,u_2)$};
        \node      at (3.1,0) {$\cdots$};
        \node[tok] (eJ) at (4.5,0) {$(c_J,u_J)$};
        \node[qtok] (cq) at (2.3,-1.0) {$c_{\text{q}}$};
        \node[draw,thick,iconblue,rounded corners,minimum width=2.6cm,minimum height=0.9cm] (T) at (2.3,-2.2) {\small Transformer $T_\theta$};
        \node[qtok] (uhat) at (2.3,-3.4) {$\hat{u}_{\text{q}}$};
        \draw[->,thick] (eJ.south) to[out=270,in=0] (T.east);

        \node[font=\scriptsize] at (2.3,0.7) {\textit{prompt of $J$ examples}};
      \end{tikzpicture}\\[0.3em]

  \end{columns}
\end{frame}

%----- Slide 3: ICON-LM proposal -----------------------------
\begin{frame}{The ICON-LM proposal: prompts \emph{plus} captions}
  \begin{block}{Two contributions over ICON (Yang et al., 2023)}
    \begin{enumerate}\setlength\itemsep{0.1em}
      \item \textbf{Multi-modal prompt}: a \emph{caption} $\kappa$ alongside the data, with knowledge that no example pair can convey.
      \item \textbf{LM-style architecture}: one transformer encoder, next-\emph{function} prediction --- structurally identical to a language model.
    \end{enumerate}
  \end{block}
  \begin{columns}[T,onlytextwidth]
    \column{0.55\textwidth}
      \textbf{Captions disclose what we know}
      \begin{itemize}\setlength\itemsep{0.1em}
        \item \emph{Vague:} ``ODE with state $u$ and control $c$;\\ $du/dt = a_1 c(t)+a_2$.'' (form only)
        \item \emph{Precise:} ``$du/dt = 0.0014\,c(t)+0.0021$.'' (parameters)
      \end{itemize}
      \vspace{0.4em}
      \textbf{Benchmark}: 19 operator families
      \begin{itemize}
        \item ODEs (\#1--8), PDEs (\#9--14), mean-field control (\#15--19).
        \item 1{,}000 operators per family, 100 condition--QoI pairs each.
      \end{itemize}
    \column{0.42\textwidth}
      \begin{tikzpicture}[scale=0.85,transform shape]
        \tikzset{
          mybox/.style={draw,rounded corners,minimum height=0.85cm,align=center,font=\scriptsize},
          capbox/.style={mybox,fill=iconaccent!20,minimum width=3.4cm},
          ex/.style ={mybox,fill=iconlight,minimum width=3.4cm},
          mod/.style={draw,thick,iconblue,rounded corners,minimum width=3.4cm,minimum height=1.0cm,fill=white},
          pred/.style={mybox,fill=iconaccent!35,minimum width=3.4cm}
        }
        \node[capbox] (cap) at (0,2.5) {\textbf{Caption}\\$du/dt=a_1 c+a_2$};
        \node[ex] (ex1) at (0,1.4) {Examples $\{(c_j,u_j)\}_{j=1}^J$};
        \node[ex] (cq) at (0,0.5) {Question condition $c_{\text{q}}$};
        \node[mod] (T) at (0,-0.7) {ICON-LM};
        \node[pred] (out) at (0,-1.8) {Prediction $\hat{u}_{\text{q}}$};
        \draw[->,thick] (cap) -- (ex1);
        \draw[->,thick] (ex1) -- (cq);

  \end{columns}
\end{frame}

\section{Architecture}

%----- Slide 4: Architecture overview -----------------------
\begin{frame}{Architecture: one transformer encoder, LM-style}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
      \textbf{From encoder--decoder ICON to ICON-LM}
      \begin{itemize}\setlength\itemsep{0.1em}
        \item Original ICON: two transformers ($\sim 31.6$M params).
        \item ICON-LM: a single GPT-style encoder ($\approx 15.8$M).
        \item Drop-in compatible with pre-trained GPT-2.
      \end{itemize}
      \vspace{0.4em}
      \textbf{Tokens}: $(\text{key},\text{value})$ pairs
      \[
        \text{key} = (\,\sigma,\ t,\ x\,),\quad \text{value} = c\ \text{or}\ u.
      \]
      \begin{itemize}\setlength\itemsep{0.1em}
        \item Three token types per example: \textbf{C}ondition, \textbf{U} (QoI), \textbf{Q}uery.
        \item Query tokens carry \emph{locations}; values zeroed out.
        \item Shared positional encoding per type $\Rightarrow$ permutation invariance within a function.
      \end{itemize}
    \column{0.48\textwidth}
      \begin{tikzpicture}[scale=0.78,transform shape]
        \tikzset{tok/.style={draw,rounded corners,minimum width=1.1cm,minimum height=0.55cm,font=\scriptsize}}
        \node[tok,fill=iconaccent!20] (cap)  at (0,0)   {CAP};
        \node[tok,fill=iconlight]      (c1) at (1.2,0) {$C_1$};
        \node[tok,fill=iconlight]      (q1) at (2.4,0) {$U_1$};
        \node[tok,fill=iconaccent!35]  (qq1) at (3.6,0) {$Q_1$};
        \node[tok,fill=iconlight]      (c2) at (4.8,0) {$C_2$};

\section{The Attention Mask}

%----- Slide 5: Custom mask (centerpiece) --------------------
\begin{frame}{The custom mask: five constraints in one diagram}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
      \textbf{Required properties}
      \begin{enumerate}\setlength\itemsep{0.1em}
        \item Predict using caption + earlier examples + current condition.
        \item \emph{No leakage} of the current QoI into its prediction.
        \item Queries within an example are mutually \emph{independent}.
        \item Token-permutation invariance within each function.
        \item No indirect leakage: $\bool(M\!\cdot\!M)=M$.
      \end{enumerate}
      \vspace{0.4em}
      \textbf{Block structure}
      \begin{itemize}\setlength\itemsep{0.05em}
        \item CAP block: lower-triangular (standard LM).
        \item $Q_j$--$Q_j$ blocks: diagonal (independent queries).
        \item $C/U$ blocks: rectangular (preserves symmetry).
      \end{itemize}
    \column{0.46\textwidth}
      \begin{tikzpicture}[scale=0.42,transform shape]
        \foreach \i in {0,...,9}{

%----- Slide 6: Training -------------------------------------
\begin{frame}{Training: next-function prediction}
  \begin{block}{Autoregressive loss across examples}
    For each $J = 0,1,\dots,I-1$ in an $I$-example prompt, predict $u_{J+1}$ from earlier examples and the caption:
    \[
      \hat{u}_{J+1} \;=\; T_\theta\!\left[c_{J+1};\,\{(c_i,u_i)\}_{i=1}^{J},\,\kappa\right].
    \]
    \[
      \mathcal{L}_{\text{prompt}} \;=\; \sum_{J=0}^{I-1}\bigl\lVert\hat{u}_{J+1}-u_{J+1}\bigr\rVert^2.
    \]
  \end{block}
  \begin{itemize}\setlength\itemsep{0.1em}
    \item One prompt simultaneously trains \emph{every} few-shot count $J$.
    \item With a caption, $J=0$ contributes (zero-shot); without, training starts at $J=1$.
    \item Multi-modal loss $=$ loss with caption $+$ loss without caption.
    \item \textbf{Inference}: append a question $c_{\text{q}}$ + query tokens, one forward pass.
  \end{itemize}
\end{frame}

\section{Discussion}

%----- Slide 9: Ablation -------------------------------------
\begin{frame}{Ablation: a tension hiding in the GPT-2 backbone}
  \begin{block}{Zero-shot relative error (precise captions, 19-family avg.)}
    \centering
    \begin{tabular}{lccc}
      \toprule
      \textbf{GPT-2 init.} & \textbf{Train} & \textbf{Test} & \textbf{Gap} \\
      \midrule
      Pre-trained          & 2.30\,\% & 2.44\,\% & \textbf{0.14\,\%} \\
      Randomly initialised & 2.07\,\% & 3.08\,\% & 1.01\,\% \\
      \bottomrule
    \end{tabular}
  \end{block}
  \begin{itemize}\setlength\itemsep{0.2em}
    \item \textbf{Pre-trained} GPT-2 generalises better across captions: language priors transfer.
    \item \textbf{Random init} wins on few-shot when many examples are given: pre-trained weights are tuned for language, not numerics.
    \item Open question: can a backbone retain both advantages simultaneously?
  \end{itemize}
\end{frame}

\section{Conclusion}

%----- Slide 11: Conclusion ----------------------------------
\begin{frame}{Conclusion}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
      \textbf{What we take away}
      \begin{itemize}\setlength\itemsep{0.2em}
        \item In-context operator learning $\to$ next-function prediction: a clean reframing as a language-model problem.
        \item Captions are a soft, transferable alternative to PINN-style constraints.
        \item Smaller \emph{and} stronger than encoder--decoder ICON; beats fine-tuned FNO/DeepONet.
      \end{itemize}
    \column{0.48\textwidth}
      \textbf{Limitations \& open questions}
      \begin{itemize}\setlength\itemsep{0.2em}
        \item Scattered-point representation: not ideal for high-D / oscillatory PDEs.
        \item Captions are GPT-4-generated: distribution shift untested.
        \item Pre-training tension: how to keep both language and numerical few-shot strengths?
      \end{itemize}
  \end{columns}
  \vspace{0.6em}
  \begin{center}
    \large\itshape Thank you \,---\, questions?
  \end{center}
\end{frame}
